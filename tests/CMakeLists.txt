cmake_minimum_required(VERSION 3.21)
project(an02062_host_tests C)

add_executable(test_ncm_ntb test_ncm_ntb.c ../app_an02062/src/ncm_ntb.c)
add_executable(test_ncm_descriptors test_ncm_descriptors.c)
option(NCM_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)

# CMake only builds native helpers. Pytest collects and runs the test cases.
foreach(target test_ncm_ntb test_ncm_descriptors)
    target_include_directories(${target} PRIVATE ../app_an02062/src)
    target_compile_features(${target} PRIVATE c_std_99)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    if(NCM_SANITIZERS)
        if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "NCM_SANITIZERS requires Clang or GCC")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
